\cvsubsection{Project 2: Finance RAG AI Assistant}

\begin{cvparagraph}
\textbf{Objective:} Create an AI-powered assistant to democratize access to finance knowledge base and reduce time spent searching documentation.
\end{cvparagraph}

\begin{cvparagraph}
\textbf{Challenge:} 300+ finance professionals spent significant time searching for guidelines, process documentation, and historical decisions. Information was scattered across SharePoint, emails, and internal wikis.
\end{cvparagraph}

\begin{cvparagraph}
\textbf{Solution:}
\begin{itemize}
    \item Developed Retrieval-Augmented Generation (RAG) system using LangChain and OpenAI GPT-4
    \item Built vector database with embeddings for 1000+ documents (guidelines, timelines, process docs)
    \item Created natural language query interface with context-aware responses
    \item Implemented source citation for traceability and compliance
    \item Deployed as web application with sub-second response times
\end{itemize}
\end{cvparagraph}

\begin{cvparagraph}
\textbf{Results:}
\begin{itemize}
    \item Reduced documentation search time by 90\%
    \item Saved 20+ hours/week across the organization
    \item Improved knowledge sharing and onboarding for new team members
    \item Zero hallucination incidents through source attribution
\end{itemize}
\end{cvparagraph}

\begin{cvparagraph}
\textbf{Technologies:} Python, LangChain, OpenAI GPT-4, Pinecone/vector DB, FastAPI, React
\end{cvparagraph}
